=====
Text block
=====
String s = """
    hello "quoted"
      indented
    """;
---
IDENT "String" @ 1:1

EQ "=" @ 1:10
TEXT_BLOCK "\"\"\"\n    hello \"quoted\"\n      indented\n    \"\"\"" @ 1:12
